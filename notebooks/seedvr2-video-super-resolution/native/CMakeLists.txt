cmake_minimum_required(VERSION 3.20)
project(seedvr2_openvino LANGUAGES CXX)

option(AI_ENHANCE_BUILD_TESTS "Build pure timing, tiling and frame invariant tests" OFF)
option(AI_ENHANCE_MATH_ONLY "Build pure tests without OpenVINO" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(seedvr2-openvino-core STATIC ai_enhance_core.cpp)
target_include_directories(seedvr2-openvino-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(seedvr2-openvino-core PRIVATE -Wall -Wextra -Wpedantic)

find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(seedvr2-openvino-assets STATIC seedvr_assets.cpp seedvr_prepare.cpp seedvr_prepare_cli.cpp)
target_include_directories(seedvr2-openvino-assets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(seedvr2-openvino-assets PUBLIC OpenSSL::Crypto)
target_compile_options(seedvr2-openvino-assets PRIVATE -Wall -Wextra -Wpedantic)

if(AI_ENHANCE_MATH_ONLY OR AI_ENHANCE_BUILD_TESTS)
  enable_testing()
  add_executable(seedvr2-openvino-core-tests ai_enhance_core_tests.cpp)
  target_link_libraries(seedvr2-openvino-core-tests PRIVATE seedvr2-openvino-core)
  target_compile_options(seedvr2-openvino-core-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME seedvr2-openvino-core COMMAND seedvr2-openvino-core-tests)
  add_executable(seedvr2-openvino-stream-tests seedvr_stream_tests.cpp)
  target_compile_options(seedvr2-openvino-stream-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME seedvr2-openvino-stream COMMAND seedvr2-openvino-stream-tests)
  add_executable(seedvr2-openvino-blob-tests seedvr_blob_tests.cpp)
  target_compile_options(seedvr2-openvino-blob-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME seedvr2-openvino-blob COMMAND seedvr2-openvino-blob-tests)
  add_executable(seedvr2-openvino-assets-tests seedvr_assets_tests.cpp)
  target_link_libraries(seedvr2-openvino-assets-tests PRIVATE seedvr2-openvino-assets)
  target_compile_options(seedvr2-openvino-assets-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(seedvr2-openvino-prepare-tests seedvr_prepare_tests.cpp)
    target_link_libraries(seedvr2-openvino-prepare-tests PRIVATE seedvr2-openvino-assets)
    target_compile_options(seedvr2-openvino-prepare-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/run-prepare-tests.cmake" CONTENT
"string(RANDOM LENGTH 16 ALPHABET 0123456789abcdef fixture)
execute_process(COMMAND \"$<TARGET_FILE:seedvr2-openvino-prepare-tests>\" \"${CMAKE_CURRENT_BINARY_DIR}/prepare-fixture-\${fixture}\" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR \"Prepare tests failed: \${result}\")
endif()
")
    add_test(NAME seedvr2-openvino-prepare COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_BINARY_DIR}/run-prepare-tests.cmake")
    add_executable(seedvr2-openvino-prepare-cli-tests seedvr_prepare_cli_tests.cpp)
    target_link_libraries(seedvr2-openvino-prepare-cli-tests PRIVATE seedvr2-openvino-assets)
    target_compile_options(seedvr2-openvino-prepare-cli-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/run-prepare-cli-tests.cmake" CONTENT
"string(RANDOM LENGTH 16 ALPHABET 0123456789abcdef fixture)
execute_process(COMMAND \"$<TARGET_FILE:seedvr2-openvino-prepare-cli-tests>\" \"${CMAKE_CURRENT_BINARY_DIR}/prepare-cli-fixture-\${fixture}\" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR \"Prepare CLI tests failed: \${result}\")
endif()
")
    add_test(NAME seedvr2-openvino-prepare-cli COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_BINARY_DIR}/run-prepare-cli-tests.cmake")
  endif()
  # Asset fixtures deliberately reject reuse; provide a fresh small leaf each run.
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/run-assets-tests.cmake" CONTENT
"string(RANDOM LENGTH 16 ALPHABET 0123456789abcdef fixture)
execute_process(COMMAND \"$<TARGET_FILE:seedvr2-openvino-assets-tests>\" \"${CMAKE_CURRENT_BINARY_DIR}/asset-fixture-\${fixture}\" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR \"Asset tests failed: \${result}\")
endif()
")
  add_test(NAME seedvr2-openvino-assets COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_BINARY_DIR}/run-assets-tests.cmake")
endif()

if(AI_ENHANCE_MATH_ONLY)
  return()
endif()

find_package(OpenVINO REQUIRED)
add_executable(seedvr2-openvino main.cpp seedvr_session.cpp seedvr_frame.cpp seedvr_geometry.cpp)
target_link_libraries(seedvr2-openvino PRIVATE seedvr2-openvino-core seedvr2-openvino-assets openvino::runtime)
target_include_directories(seedvr2-openvino PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(seedvr2-openvino PRIVATE -Wall -Wextra -Wpedantic)

set(AI_ENHANCE_RUNTIME_TARGETS seedvr2-openvino)
if(AI_ENHANCE_BUILD_TESTS)
  add_executable(seedvr2-openvino-cli-tests seedvr_cli_tests.cpp seedvr_frame.cpp seedvr_session.cpp seedvr_geometry.cpp)
  target_link_libraries(seedvr2-openvino-cli-tests PRIVATE seedvr2-openvino-core seedvr2-openvino-assets openvino::runtime)
  target_compile_options(seedvr2-openvino-cli-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME seedvr2-openvino-cli COMMAND seedvr2-openvino-cli-tests)
  add_executable(seedvr2-openvino-frame-tests seedvr_frame_tests.cpp seedvr_frame.cpp seedvr_session.cpp seedvr_geometry.cpp)
  target_link_libraries(seedvr2-openvino-frame-tests PRIVATE seedvr2-openvino-core seedvr2-openvino-assets openvino::runtime)
  target_compile_options(seedvr2-openvino-frame-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME seedvr2-openvino-frame COMMAND seedvr2-openvino-frame-tests)
  add_executable(seedvr2-openvino-geometry-tests seedvr_geometry_tests.cpp seedvr_geometry.cpp)
  target_link_libraries(seedvr2-openvino-geometry-tests PRIVATE seedvr2-openvino-core openvino::runtime)
  target_compile_options(seedvr2-openvino-geometry-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME seedvr2-openvino-geometry COMMAND seedvr2-openvino-geometry-tests)
  add_executable(seedvr2-openvino-blob-fallback-tests seedvr_blob_fallback_tests.cpp)
  target_link_libraries(seedvr2-openvino-blob-fallback-tests PRIVATE openvino::runtime)
  target_compile_options(seedvr2-openvino-blob-fallback-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME seedvr2-openvino-blob-fallback COMMAND seedvr2-openvino-blob-fallback-tests)
  list(APPEND AI_ENHANCE_RUNTIME_TARGETS seedvr2-openvino-cli-tests seedvr2-openvino-frame-tests seedvr2-openvino-geometry-tests
    seedvr2-openvino-blob-fallback-tests)
endif()

get_target_property(OPENVINO_RUNTIME_LIBRARY openvino::runtime IMPORTED_LOCATION_RELEASE)
if(NOT OPENVINO_RUNTIME_LIBRARY)
  get_target_property(OPENVINO_RUNTIME_LIBRARY openvino::runtime IMPORTED_LOCATION)
endif()
if(OPENVINO_RUNTIME_LIBRARY)
  get_filename_component(OPENVINO_RUNTIME_LIB_DIR "${OPENVINO_RUNTIME_LIBRARY}" DIRECTORY)
  get_filename_component(OPENVINO_RUNTIME_ROOT "${OPENVINO_RUNTIME_LIB_DIR}/../.." ABSOLUTE)
  find_file(AI_ENHANCE_TBB_LIBRARY NAMES libtbb.so.12.13 libtbb.so.12
    PATHS "${OPENVINO_RUNTIME_LIB_DIR}" "${OPENVINO_RUNTIME_ROOT}/3rdparty/tbb/lib" NO_DEFAULT_PATH)
  if(AI_ENHANCE_TBB_LIBRARY)
    get_filename_component(AI_ENHANCE_TBB_LIB_DIR "${AI_ENHANCE_TBB_LIBRARY}" DIRECTORY)
    foreach(target IN LISTS AI_ENHANCE_RUNTIME_TARGETS)
      target_link_libraries(${target} PRIVATE "${AI_ENHANCE_TBB_LIBRARY}")
      target_link_options(${target} PRIVATE "-Wl,-rpath-link,${AI_ENHANCE_TBB_LIB_DIR}")
    endforeach()
  endif()
endif()
set_target_properties(seedvr2-openvino PROPERTIES
  BUILD_RPATH "$ORIGIN/lib"
  INSTALL_RPATH "$ORIGIN/lib"
  BUILD_WITH_INSTALL_RPATH TRUE
)
